\begin{document}

\begin{tikzpicture}[
    >=Stealth,
    line width=0.6pt,
    every node/.style={font=\small},
    stage/.style={
        draw,
        rounded corners=2pt,
        minimum height=2.7cm,
        minimum width=1.7cm,
        align=center
    },
    feature/.style={
        draw,
        minimum width=1.35cm,
        minimum height=1.35cm
    },
    arrow/.style={->, line width=0.75pt}
]

% Input image / signal
\node[stage, minimum width=1.9cm] (input) at (0,0) {};
\node[above=3pt of input] {Input};
\draw[step=0.38cm, gray!55, thin] ($(input.south west)+(0.2,0.2)$)
    grid ($(input.north east)+(-0.2,-0.2)$);
\node[below=5pt of input, align=center] {Image, feature map,\\or sensor data};

% Convolution feature maps
\node[feature] (conv1) at (3.4,0.75) {};
\node[feature] (conv2) at (3.7,0.35) {};
\node[feature] (conv3) at (4.0,-0.05) {};
\foreach \n/\shift in {conv1/0,conv2/-0.4,conv3/-0.8} {
  \draw[step=0.27cm, gray!55, thin]
    ($ (\n.south west)+(0.14,0.14) $) grid
    ($ (\n.north east)+(-0.14,-0.14) $);
}
\node[above=0.55cm of conv1] {Convolution};
\node[below=1.35cm of conv2, align=center] {Learned local features\\(filters / kernels)};

% ReLU
\node[stage, minimum width=1.65cm] (relu) at (6.8,0) {ReLU\\activation};
\node[below=5pt of relu, align=center] {$\operatorname{ReLU}(z)=\max(0,z)$};

% Pooling
\node[stage, minimum width=1.7cm] (pool) at (9.5,0) {Pooling\\/ downsampling};
\draw[step=0.32cm, gray!55, thin] ($(pool.south west)+(0.35,0.5)$)
    grid ($(pool.north east)+(-0.35,-0.5)$);
\node[below=5pt of pool, align=center] {Reduced spatial\\resolution};

% Fully connected
\node[stage, minimum width=1.8cm] (fc) at (12.3,0) {Fully connected\\layer};
\foreach \y in {-0.55,0,0.55} {
  \fill ($(fc.center)+(-0.45,\y)$) circle (1.4pt);
}
\foreach \y in {-0.38,0.38} {
  \fill ($(fc.center)+(0.45,\y)$) circle (1.4pt);
}
\foreach \ya in {-0.55,0,0.55} {
  \foreach \yb in {-0.38,0.38} {
    \draw[gray!70, thin] ($(fc.center)+(-0.45,\ya)$) -- ($(fc.center)+(0.45,\yb)$);
  }
}

% Output
\node[stage, minimum width=1.6cm] (output) at (15.1,0) {Output};
\node[below=5pt of output, align=center] {Class scores,\\regression value,\\or control action};

% Main arrows
\draw[arrow] (input.east) -- (conv1.west);
\draw[arrow] (conv3.east) -- (relu.west);
\draw[arrow] (relu.east) -- (pool.west);
\draw[arrow] (pool.east) -- (fc.west);
\draw[arrow] (fc.east) -- (output.west);

% Repetition bracket
\draw[decorate, decoration={brace, amplitude=4pt, mirror}]
  ($(conv2.south west)+(-0.1,-1.65)$) -- ($(pool.south east)+(0.1,-1.65)$)
  node[midway, below=6pt, align=center] {Repeated convolution, activation,\\and optional pooling stages};

\end{tikzpicture}
